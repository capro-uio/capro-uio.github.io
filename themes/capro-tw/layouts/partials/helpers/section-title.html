{{- /* Section heading: first word plain, remainder accented. Takes either a plain title string, or a dict {title, page}. When a page is supplied and has no

yet (list pages have no hero), the first section title on that page is promoted to

so every page has one. The home page gets its

from the banner. The two levels are written out in full rather than interpolating the tag name: Go's html/template cannot compute an output context for a dynamic element name and silently renders nothing. */ -}} {{- $title := . -}} {{- $page := false -}} {{- if reflect.IsMap . -}} {{- $title = .title -}} {{- $page = .page -}} {{- end -}} {{- with $title -}} {{- $promote := false -}} {{- if and $page (not $page.IsHome) -}} {{- if not ($page.Scratch.Get "sectionH1Used") -}} {{- $promote = true -}} {{- $page.Scratch.Set "sectionH1Used" true -}} {{- end -}} {{- end -}} {{- if $page -}}{{- $page.Scratch.Set "itemLevel" (cond $promote 2 3) -}}{{- end -}} {{- $words := split . " " -}} {{- $head := index $words 0 -}} {{- $rest := "" -}} {{- if gt (len $words) 1 -}}{{- $rest = delimit (after 1 $words) " " -}}{{- end -}}
{{ if $promote -}}

{{ $head | safeHTML }}{{ with $rest }} {{ . | safeHTML }}{{ end }}

{{- else -}}

{{ $head | safeHTML }}{{ with $rest }} {{ . | safeHTML }}{{ end }}

{{- end }}
{{- end -}}
